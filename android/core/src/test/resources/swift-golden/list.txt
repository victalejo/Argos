for d in "$HOME/.config/argos/agents"/*/; do [ -f "$d/meta.json" ] || continue; id=$(basename "$d"); p=$(cat "$d/pid" 2>/dev/null); alive=0; [ -n "$p" ] && ps -o args= -p "$p" 2>/dev/null | grep -q -- "$id" && alive=1; size=$(wc -c < "$d/out.ndjson" 2>/dev/null | tr -d ' '); b=$(cat "$d/log.base" 2>/dev/null); case "$b" in ''|*[!0-9]*) b=0;; esac; size=$(( ${size:-0} + b )); printf '%s|%s|%s|' "$id" "$alive" "$size"; base64 < "$d/meta.json" | tr -d '\n'; echo; done
