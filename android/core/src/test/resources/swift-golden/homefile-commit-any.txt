umask 077
f="$HOME/.claude/settings.json"; t="$f.argos-tmp-0123456789abcdef0123456789abcdef"; l="$f.argos-lock"
n=0
until mkdir "$l" 2>/dev/null; do
  if [ -n "$(find "$l" -maxdepth 0 -mmin +1 2>/dev/null)" ]; then rmdir "$l" 2>/dev/null; continue; fi
  n=$((n + 1)); if [ "$n" -ge 10 ]; then rm -f -- "$t"; exit 76; fi
  sleep 1
done
trap 'rmdir "$l" 2>/dev/null' EXIT
if [ -L "$f" ]; then rm -f -- "$t"; exit 77; fi
if [ -f "$f" ]; then rm -f -- "$f.argos-backup" && cp -- "$f" "$f.argos-backup" || { rm -f -- "$t"; exit 1; }; fi
mv -f -- "$t" "$f" || { rm -f -- "$t"; exit 1; }
find "${f%/*}" -maxdepth 1 -name 'settings.json.argos-tmp-*' -mmin +60 -exec rm -f -- {} + 2>/dev/null
exit 0
