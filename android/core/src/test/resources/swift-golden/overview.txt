now=$(date +%s); for d in "$HOME/.config/argos/agents"/*/; do [ -f "$d/meta.json" ] || continue; id=$(basename "$d"); p=$(cat "$d/pid" 2>/dev/null); alive=0; [ -n "$p" ] && ps -o args= -p "$p" 2>/dev/null | grep -q -- "$id" && alive=1; size=$(wc -c < "$d/out.ndjson" 2>/dev/null | tr -d ' '); b=$(cat "$d/log.base" 2>/dev/null); case "$b" in ''|*[!0-9]*) b=0;; esac; size=$(( ${size:-0} + b )); m=$(stat -c %Y "$d/out.ndjson" 2>/dev/null || stat -f %m "$d/out.ndjson" 2>/dev/null); idle=$(( now - ${m:-$now} )); printf '%s|%s|%s|%s|' "$id" "$alive" "$size" "$idle"; base64 < "$d/meta.json" | tr -d '\n'; printf '|'; tail -n 30 "$d/out.ndjson" 2>/dev/null | sed -E '/"tool_name":"/ s/^(.{1,200}).*("tool_name":"[^"]*").*$/\1 \2/' | cut -c 1-400 | base64 | tr -d '\n'; echo; done; true
