tmux list-panes -a -F '#{@argos_claude}|#{pane_id}|#{window_index}|#{session_name}' 2>/dev/null | while IFS= read -r l; do s=${l%%[|]*}; r=${l#*[|]}; p=${r%%[|]*}; c=-; if [ "$s" = working ] || [ "$s" = waiting ]; then t=$(tmux capture-pane -p -t "$p" 2>/dev/null | grep -v '^[[:space:]]*$'); c=0; printf '%s\n' "$t" | grep -q '… ([0-9]' && c=1; printf '%s\n' "$t" | tail -n 4 | grep -q 'esc to interrupt' && c=2; fi; printf '%s|%s\n' "$c" "$l"; done
